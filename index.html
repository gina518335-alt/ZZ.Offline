async function searchAddr() {

  const raw = document.getElementById('search-input').value.trim();

  if(!raw) return;

  const btn = document.getElementById('search-btn');

  btn.disabled=true; btn.textContent='...';

  setSS('검색 중...');

  document.getElementById('search-results').innerHTML='';

  const isPostal = /^\d{5}$/.test(raw);

  const queries = isPostal

    ? [`https://nominatim.openstreetmap.org/search?postalcode=${raw}&countrycodes=kr&format=json&limit=5&accept-language=ko`]

    : [

        `https://nominatim.openstreetmap.org/search?q=${encodeURIComponent(raw)}&countrycodes=kr&format=json&limit=5&accept-language=ko`,

        `https://nominatim.openstreetmap.org/search?q=${encodeURIComponent(raw+' 서울')}&countrycodes=kr&format=json&limit=5&accept-language=ko`,

      ];

  let data = [];

  for (const url of queries) {

    try {

      const res = await fetch(url);

      data = await res.json();

      if (data.length) break;

    } catch(e) {}

  }

  if(!data.length){

    setSS('결과가 없어요. 도로명 주소나 우편번호(5자리)로 시도해 보세요.');

    btn.disabled=false; btn.textContent='검색'; return;

  }

  setSS(isPostal ? `우편번호 ${raw} 결과` : '');

  const wrap=document.getElementById('search-results');

  data.forEach(r=>{

    const div=document.createElement('div'); div.className='search-result-item';

    const parts = r.display_name.split(',');

    const name = parts[0].trim();

    const addr = parts.slice(1,4).filter(Boolean).join(', ').trim();

    div.innerHTML=`<div class="res-name">${name}</div><div class="res-addr">${addr}</div>`;

    div.onclick=()=>{

      pendingLatLng={lat:parseFloat(r.lat),lng:parseFloat(r.lon)};

      map.flyTo([pendingLatLng.lat,pendingLatLng.lng],17,{duration:0.7});

      document.getElementById('search-results').innerHTML='';

      document.getElementById('search-input').value=name;

      setSS('');

      document.getElementById('f-name').value=name;

      setTimeout(openModal,600);

    };

    wrap.appendChild(div);

  });

  btn.disabled=false; btn.textContent='검색';

}
